CREATE TABLE IF NOT EXISTS email_jobs (
    email_id int4 GENERATED BY DEFAULT AS IDENTITY,

    email_from_id varchar(100) NULL,
    email_to_id varchar(250) NULL,
    email_cc_id varchar(250) NULL,
    email_bcc_id varchar(250) NULL,
    email_subject varchar(100) NULL,
    email_message varchar(250) NULL,
    email_file_folder varchar(100) NULL,
    email_ctr int4 NULL,

    email_scheduled_on timestamptz NULL,
    email_send_date timestamptz NULL,

    email_status varchar(15) NULL,
    email_error_msg varchar(100) NULL,
    email_remarks varchar(100) NULL,

    rec_created_date timestamptz NULL,
    rec_created_by varchar(15) NULL,
    rec_company_id int4 NOT NULL,
    rec_branch_id int4  NOT NULL,

    CONSTRAINT pk_email_jobs_email_id PRIMARY KEY (email_id)
);

DO $$
BEGIN
    IF NOT EXISTS (SELECT 1 FROM pg_indexes WHERE indexname = 'IX_email_jobs_rec_company_id') THEN
        CREATE INDEX "IX_email_jobs_rec_company_id" ON email_jobs USING btree (rec_company_id);
    END IF;

    IF NOT EXISTS (SELECT 1 FROM pg_indexes WHERE indexname = 'IX_email_jobs_rec_branch_id') THEN
        CREATE INDEX "IX_email_jobs_rec_branch_id" ON email_jobs USING btree (rec_branch_id);
    END IF;

END $$;

SELECT setval(
    pg_get_serial_sequence('email_jobs', 'email_id'),
    COALESCE(MAX(email_id), 1),
    true
)
FROM email_jobs;
