CREATE TABLE IF NOT EXISTS email_list (
    email_id int4 GENERATED BY DEFAULT AS IDENTITY,

    email_type varchar(50) NULL,
    email_from_id varchar(100) NULL,
    email_to_id varchar(250) NULL,
    email_cc_id varchar(250) NULL,
    email_bcc_id varchar(250) NULL,
    email_subject varchar(100) NULL,
    email_message varchar(250) NULL,

    rec_created_date timestamptz NULL,
    rec_created_by varchar(15) NULL,
    rec_company_id int4 NOT NULL,
    rec_branch_id int4  NOT NULL,

    CONSTRAINT pk_email_list_email_id PRIMARY KEY (email_id)
);

DO $$
BEGIN
    IF NOT EXISTS (SELECT 1 FROM pg_indexes WHERE indexname = 'IX_email_list_rec_company_id') THEN
        CREATE INDEX "IX_email_list_rec_company_id" ON email_list USING btree (rec_company_id);
    END IF;

    IF NOT EXISTS (SELECT 1 FROM pg_indexes WHERE indexname = 'IX_email_list_rec_branch_id') THEN
        CREATE INDEX "IX_email_list_rec_branch_id" ON email_list USING btree (rec_branch_id);
    END IF;
END $$;

SELECT setval(
    pg_get_serial_sequence('email_list', 'email_id'),
    COALESCE(MAX(email_id), 1),
    true
)
FROM email_list;
