CREATE TABLE IF NOT EXISTS cargo_approvedd (
    cad_id int4 GENERATED BY DEFAULT AS IDENTITY,

    cad_parent_id int4 NULL,
    cad_approved_by varchar(30) NULL,
    cad_is_approved boolean NULL,
    cad_approved_date timestamptz NULL,
    cad_is_hide boolean NULL,

    rec_record_id int4 NULL,
    rec_location_id int4 NULL,
    rec_locked boolean NULL,
    rec_closed boolean NULL,
    rec_created_date timestamptz NULL,
    rec_edited_date timestamptz NULL,
    rec_created_by varchar(50) NULL,
    rec_edited_by varchar(50) NULL,
    rec_company_id int4 NULL,
    rec_branch_id int4 NULL,

    CONSTRAINT pk_cargo_approvedd_cad_id PRIMARY KEY (cad_id)
);

DO $$
BEGIN
    IF NOT EXISTS ( SELECT 1 FROM pg_indexes WHERE indexname = 'ix_cargo_approvedd_rec_record_id' ) THEN
        CREATE INDEX "ix_cargo_approvedd_rec_record_id" ON cargo_approvedd USING btree (rec_record_id);
    END IF;

    IF NOT EXISTS ( SELECT 1 FROM pg_indexes WHERE indexname = 'ix_cargo_approvedd_rec_location_id' ) THEN
        CREATE INDEX "ix_cargo_approvedd_rec_location_id" ON cargo_approvedd USING btree (rec_location_id);
    END IF;

    IF NOT EXISTS (
        SELECT 1
        FROM pg_indexes
        WHERE indexname = 'ix_cargo_approvedd_ca_ref_id'
    ) THEN
        CREATE INDEX "ix_cargo_approvedd_ca_ref_id"
        ON cargo_approvedd USING btree (ca_ref_id);
    END IF;

    IF NOT EXISTS (
        SELECT 1
        FROM pg_indexes
        WHERE indexname = 'ix_cargo_approvedd_ca_hbl_id'
    ) THEN
        CREATE INDEX "ix_cargo_approvedd_ca_hbl_id"
        ON cargo_approvedd USING btree (ca_hbl_id);
    END IF;

    IF NOT EXISTS (
        SELECT 1
        FROM pg_indexes
        WHERE indexname = 'ix_cargo_approvedd_ca_inv_id'
    ) THEN
        CREATE INDEX "ix_cargo_approvedd_ca_inv_id"
        ON cargo_approvedd USING btree (ca_inv_id);
    END IF;
END $$;

SELECT setval(
    pg_get_serial_sequence('cargo_approvedd', 'cad_id'),
    COALESCE(MAX(cad_id), 1),
    true
)
FROM cargo_approvedd;
