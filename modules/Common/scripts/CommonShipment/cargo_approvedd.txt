CREATE TABLE IF NOT EXISTS cargo_approvedd (
    cad_id int4 GENERATED BY DEFAULT AS IDENTITY,

    cad_parent_id int4 NULL,
    cad_approvedby_id int4 NULL,
    cad_is_approved varchar(5) NULL,
    cad_approved_date timestamptz NULL,
    cad_is_hide varchar(5) NULL,

    rec_created_date timestamptz NULL,
    rec_edited_date timestamptz NULL,
    rec_created_by varchar(50) NULL,
    rec_edited_by varchar(50) NULL,
    rec_company_id int4 NOT NULL,
    rec_branch_id int4 NOT NULL,

    CONSTRAINT pk_cargo_approvedd_cad_id PRIMARY KEY (cad_id)
);

DO $$
BEGIN
    IF NOT EXISTS ( SELECT 1 FROM pg_indexes WHERE indexname = 'ix_cargo_approvedd_cad_parent_id' ) THEN
        CREATE INDEX "ix_cargo_approvedd_cad_parent_id" ON cargo_approvedd USING btree (cad_parent_id);
    END IF;
    
    IF NOT EXISTS ( SELECT 1 FROM pg_indexes WHERE indexname = 'ix_cargo_approvedd_cad_approvedby_id' ) THEN
        CREATE INDEX "ix_cargo_approvedd_cad_approvedby_id" ON cargo_approvedd USING btree (cad_approvedby_id);
    END IF;

    IF NOT EXISTS (SELECT 1 FROM pg_indexes WHERE indexname = 'IX_cargo_approvedd_rec_company_id') THEN
        CREATE INDEX "IX_cargo_approvedd_rec_company_id" ON cargo_approvedd USING btree (rec_company_id);
    END IF;

    IF NOT EXISTS (SELECT 1 FROM pg_indexes WHERE indexname = 'IX_cargo_approvedd_rec_branch_id') THEN
        CREATE INDEX "IX_cargo_approvedd_rec_branch_id" ON cargo_approvedd USING btree (rec_branch_id);
    END IF;
END $$;

SELECT setval(
    pg_get_serial_sequence('cargo_approvedd', 'cad_id'),
    COALESCE(MAX(cad_id), 1),
    true
)
FROM cargo_approvedd;
