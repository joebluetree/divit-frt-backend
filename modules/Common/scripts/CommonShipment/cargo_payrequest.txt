CREATE TABLE IF NOT EXISTS cargo_payrequest (
    cp_id int4 GENERATED BY DEFAULT AS IDENTITY,

    cp_slno int4 NOT NULL,
    cp_mode varchar(50) NULL,
    cp_source varchar(50) NULL,
    cp_master_id int4 NULL,
    cp_paytype_needed varchar(30) NULL,
    cp_spl_notes varchar(100) NULL,
    cp_payment_date date NULL,
    cp_pay_status varchar(50) NULL,
    cp_cust_id int4 NULL,
    cp_inv_id int4 NULL,
    cp_inv_no varchar(20) NULL,

    rec_version int4 DEFAULT 1 NOT NULL,
    rec_deleted varchar(5) NULL,
    rec_deleted_by varchar(30) NULL,
    rec_deleted_date date NULL,
    rec_record_id int4 NULL,
    rec_location_id int4 NULL,
    rec_locked varchar(5) NULL,
    rec_closed varchar(5) NULL,
    rec_created_date timestamptz NULL,
    rec_edited_date timestamptz NULL,
    rec_created_by varchar(50) NULL,
    rec_edited_by varchar(50) NULL,
    rec_company_id int4 NOT NULL,
    rec_branch_id int4 NOT NULL,
    rec_created_id int4 NULL,

    CONSTRAINT pk_cargo_payrequest_cp_id PRIMARY KEY (cp_id)
);

DO $$
BEGIN
    IF NOT EXISTS (SELECT 1 FROM pg_indexes WHERE indexname = 'ix_cargo_payrequest_cp_master_id') THEN
        CREATE INDEX "ix_cargo_payrequest_cp_master_id" ON cargo_payrequest USING btree (cp_master_id);
    END IF;

    IF NOT EXISTS (SELECT 1 FROM pg_indexes WHERE indexname = 'ix_cargo_payrequest_cp_cust_id') THEN
        CREATE INDEX "ix_cargo_payrequest_cp_cust_id" ON cargo_payrequest USING btree (cp_cust_id);
    END IF;

    IF NOT EXISTS (SELECT 1 FROM pg_indexes WHERE indexname = 'ix_cargo_payrequest_cp_inv_id') THEN
        CREATE INDEX "ix_cargo_payrequest_cp_inv_id" ON cargo_payrequest USING btree (cp_inv_id);
    END IF;

    IF NOT EXISTS (SELECT 1 FROM pg_indexes WHERE indexname = 'ix_cargo_payrequest_rec_record_id') THEN
        CREATE INDEX "ix_cargo_payrequest_rec_record_id" ON cargo_payrequest USING btree (rec_record_id);
    END IF;

    IF NOT EXISTS (SELECT 1 FROM pg_indexes WHERE indexname = 'ix_cargo_payrequest_rec_location_id') THEN
        CREATE INDEX "ix_cargo_payrequest_rec_location_id" ON cargo_payrequest USING btree (rec_location_id);
    END IF;

    IF NOT EXISTS (SELECT 1 FROM pg_indexes WHERE indexname = 'IX_cargo_payrequest_rec_company_id') THEN
        CREATE INDEX "IX_cargo_payrequest_rec_company_id" ON cargo_payrequest USING btree (rec_company_id);
    END IF;

    IF NOT EXISTS (SELECT 1 FROM pg_indexes WHERE indexname = 'IX_cargo_payrequest_rec_branch_id') THEN
        CREATE INDEX "IX_cargo_payrequest_rec_branch_id" ON cargo_payrequest USING btree (rec_branch_id);
    END IF;

    IF NOT EXISTS (SELECT 1 FROM pg_indexes WHERE indexname = 'IX_cargo_payrequest_rec_created_id') THEN
        CREATE INDEX "IX_cargo_payrequest_rec_created_id" ON cargo_payrequest USING btree (rec_created_id);
    END IF;
END $$;

SELECT setval(
    pg_get_serial_sequence('cargo_payrequest', 'cp_id'),
    COALESCE(MAX(cp_id), 1),
    true
)
FROM cargo_payrequest;
