CREATE TABLE IF NOT EXISTS cargo_approvedm (
    ca_id int4 GENERATED BY DEFAULT AS IDENTITY,

    ca_type varchar(50) NULL,
    ca_doc_type varchar(50) NULL,
    ca_ref_id int4 NULL,
    ca_ref_no varchar(100) NULL,
    ca_remarks text NULL,
    ca_user_id varchar(50) NULL,
    ca_is_approved boolean NULL,
    ca_date timestamptz NULL,

    ca_hbl_id int4 NULL,
    ca_inv_id int4 NULL,
    ca_consignee_id int4 NULL,

    ca_approved_tot numeric(18,2) NULL,
    ca_notapproved_tot numeric(18,2) NULL,

    ca_obl_recvd_date timestamptz NULL,
    ca_payment_recvd_date timestamptz NULL,

    ca_is_hide boolean NULL,
    ca_req_no varchar(100) NULL,
    ca_is_ar_issued boolean NULL,

    rec_files_attached int4 NULL,
    ca_is_hide2 boolean NULL,

    rec_record_id int4 NULL,
    rec_location_id int4 NULL,
    rec_locked boolean NULL,
    rec_closed boolean NULL,
    rec_created_date timestamptz NULL,
    rec_edited_date timestamptz NULL,
    rec_created_by varchar(50) NULL,
    rec_edited_by varchar(50) NULL,
    rec_company_id int4 NULL,
    rec_branch_id int4 NULL,

    CONSTRAINT pk_cargo_approvedm_ca_pkid PRIMARY KEY (ca_pkid)
);

DO $$
BEGIN
    IF NOT EXISTS (
        SELECT 1
        FROM pg_indexes
        WHERE indexname = 'ix_cargo_approvedm_rec_record_id'
    ) THEN
        CREATE INDEX "ix_cargo_approvedm_rec_record_id"
        ON cargo_approvedm USING btree (rec_record_id);
    END IF;

    IF NOT EXISTS (
        SELECT 1
        FROM pg_indexes
        WHERE indexname = 'ix_cargo_approvedm_rec_location_id'
    ) THEN
        CREATE INDEX "ix_cargo_approvedm_rec_location_id"
        ON cargo_approvedm USING btree (rec_location_id);
    END IF;

    IF NOT EXISTS (
        SELECT 1
        FROM pg_indexes
        WHERE indexname = 'ix_cargo_approvedm_ca_ref_id'
    ) THEN
        CREATE INDEX "ix_cargo_approvedm_ca_ref_id"
        ON cargo_approvedm USING btree (ca_ref_id);
    END IF;

    IF NOT EXISTS (
        SELECT 1
        FROM pg_indexes
        WHERE indexname = 'ix_cargo_approvedm_ca_hbl_id'
    ) THEN
        CREATE INDEX "ix_cargo_approvedm_ca_hbl_id"
        ON cargo_approvedm USING btree (ca_hbl_id);
    END IF;

    IF NOT EXISTS (
        SELECT 1
        FROM pg_indexes
        WHERE indexname = 'ix_cargo_approvedm_ca_inv_id'
    ) THEN
        CREATE INDEX "ix_cargo_approvedm_ca_inv_id"
        ON cargo_approvedm USING btree (ca_inv_id);
    END IF;
END $$;

SELECT setval(
    pg_get_serial_sequence('cargo_approvedm', 'ca_pkid'),
    COALESCE(MAX(ca_pkid), 1),
    true
)
FROM cargo_approvedm;
