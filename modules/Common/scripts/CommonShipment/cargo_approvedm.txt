CREATE TABLE IF NOT EXISTS cargo_approvedm (
    ca_id int4 GENERATED BY DEFAULT AS IDENTITY,

    ca_req_no int4 NOT NULL,
    ca_type varchar(50) NULL,
    ca_doc_type varchar(50) NULL,
    ca_mbl_id int4 NULL,
    ca_ref_no varchar(30) NULL,
    ca_remarks varchar(100) NULL,
    ca_user_id int4 NULL,
    ca_is_approved varchar(5) NULL,
    ca_date date NULL,

    ca_hbl_id int4 NULL,
    ca_inv_id int4 NULL,
    ca_consignee_id int4 NULL,
    ca_approved_tot numeric(15,3) NULL,
    ca_notapproved_tot numeric(15,3) NULL,
    ca_obl_recvd_date timestamptz NULL,
    ca_payment_recvd_date timestamptz NULL,
    ca_is_hide varchar(5) NULL,
    ca_is_ar_issued varchar(5) NULL,
    rec_files_attached varchar(5) NULL,
    ca_is_hide2 varchar(5) NULL,

    rec_record_id int4 NULL,
    rec_location_id int4 NULL,
    rec_locked varchar(5) NULL,
    rec_closed varchar(5) NULL,
    rec_created_date timestamptz NULL,
    rec_edited_date timestamptz NULL,
    rec_created_by varchar(50) NULL,
    rec_edited_by varchar(50) NULL,
    rec_company_id int4 NOT NULL,
    rec_branch_id int4 NOT NULL,
    rec_created_id int4 NULL,

    CONSTRAINT pk_cargo_approvedm_ca_id PRIMARY KEY (ca_id)
);

DO $$
BEGIN
    IF NOT EXISTS (SELECT 1 FROM pg_indexes WHERE indexname = 'ix_cargo_approvedm_ca_mbl_id') THEN
        CREATE INDEX "ix_cargo_approvedm_ca_mbl_id" ON cargo_approvedm USING btree (ca_mbl_id);
    END IF;

    IF NOT EXISTS (SELECT 1 FROM pg_indexes WHERE indexname = 'ix_cargo_approvedm_ca_user_id') THEN
        CREATE INDEX "ix_cargo_approvedm_ca_user_id" ON cargo_approvedm USING btree (ca_user_id);
    END IF;

    IF NOT EXISTS (SELECT 1 FROM pg_indexes WHERE indexname = 'ix_cargo_approvedm_ca_hbl_id') THEN
        CREATE INDEX "ix_cargo_approvedm_ca_hbl_id" ON cargo_approvedm USING btree (ca_hbl_id);
    END IF;

    IF NOT EXISTS (SELECT 1 FROM pg_indexes WHERE indexname = 'ix_cargo_approvedm_ca_inv_id') THEN
        CREATE INDEX "ix_cargo_approvedm_ca_inv_id" ON cargo_approvedm USING btree (ca_inv_id);
    END IF;

    IF NOT EXISTS (SELECT 1 FROM pg_indexes WHERE indexname = 'ix_cargo_approvedm_ca_consignee_id') THEN
        CREATE INDEX "ix_cargo_approvedm_ca_consignee_id" ON cargo_approvedm USING btree (ca_consignee_id);
    END IF;

    IF NOT EXISTS (SELECT 1 FROM pg_indexes WHERE indexname = 'ix_cargo_approvedm_rec_record_id') THEN
        CREATE INDEX "ix_cargo_approvedm_rec_record_id" ON cargo_approvedm USING btree (rec_record_id);
    END IF;

    IF NOT EXISTS (SELECT 1 FROM pg_indexes WHERE indexname = 'ix_cargo_approvedm_rec_location_id') THEN
        CREATE INDEX "ix_cargo_approvedm_rec_location_id" ON cargo_approvedm USING btree (rec_location_id);
    END IF;

    IF NOT EXISTS (SELECT 1 FROM pg_indexes WHERE indexname = 'IX_cargo_approvedm_rec_company_id') THEN
        CREATE INDEX "IX_cargo_approvedm_rec_company_id" ON cargo_approvedm USING btree (rec_company_id);
    END IF;

    IF NOT EXISTS (SELECT 1 FROM pg_indexes WHERE indexname = 'IX_cargo_approvedm_rec_branch_id') THEN
        CREATE INDEX "IX_cargo_approvedm_rec_branch_id" ON cargo_approvedm USING btree (rec_branch_id);
    END IF;

    IF NOT EXISTS (SELECT 1 FROM pg_indexes WHERE indexname = 'IX_cargo_approvedm_rec_created_id') THEN
        CREATE INDEX "IX_cargo_approvedm_rec_created_id" ON cargo_approvedm USING btree (rec_created_id);
    END IF;
END $$;

SELECT setval(
    pg_get_serial_sequence('cargo_approvedm', 'ca_id'),
    COALESCE(MAX(ca_id), 1),
    true
)
FROM cargo_approvedm;
